სს საქართველოს ენერგეტიკის განვითარების ფონდისათვის საქონლისა და მომსახურების შესყიდვების სპეციალური წესი
მუხლი 1. ზოგადი დებულებები
1. "სს საქართველოს ენერგეტიკის განვითარების ფონდისათვის საქონლისა და მომსახურების შესყიდვების სპეციალური წესი" (შემდგომში - წესი) შემუშავებულია "სახელმწიფო შესყიდვების შესახებ" საქართველოს კანონის მე-3 მუხლის პირველი პუნქტის "ა.თ" ქვეპუნქტის შესაბამისად და ადგენს ამ წესით განსაზღვრული საქონლისა და მომსახურების შესყიდვების წესს.
2. ამ წესის მიზანია განსახორციელებელი მომსახურებისა და საქონლის შესყიდვების პროცედურების გამარტივება და შემსყიდველის მიზნების შეუფერხებლად განხორციელების უზრუნველყოფა.
მუხლი 2. ტერმინთა განმარტება
ამ წესში გამოყენებულ ტერმინებს აქვს შემდეგი მნიშვნელობა:
ა) შემსყიდველი - სს საქართველოს ენერგეტიკის განვითარების ფონდი;
ბ) შესყიდვა - სს საქართველოს ენერგეტიკის განვითარების ფონდის მიერ იმ საქონლისა და მომსახურების შესყიდვა, რომლებიც მოცემულია ამ წესის №1 და №2 დანართებში;
გ) პირდაპირი შესყიდვა - შესყიდვის საშუალება, რომელიც გამოიყენება ამ წესის მე-4 მუხლით გათვალისწინებულ შემთხვევებში;
დ) სახელმწიფო შესყიდვების ერთიანი ელექტრონული სისტემა (შემდგომში - სისტემა) - საქართველოს კანონმდებლობით გათვალისწინებული სახელმწიფო შესყიდვების ოფიციალური პორტალი https://tenders.procurement.gov.ge;
ე) შესყიდვის ელექტრონული პროცედურა - ამ წესის დანართით გათვალისწინებული საქონლისა და მომსახურების შესყიდვის საშუალება, რომლის განხორციელების წესი და პირობები განისაზღვრება ამ წესით (ამ წესით განსაზღვრული ხელშეკრულების შესრულების უზრუნველყოფის გარანტიის მიზნებისათვის შესაძლებელია გამოყენებულ იქნეს ტერმინი - ტენდერი);
ვ) შესყიდვის ელექტრონული პროცედურის პირობები (შემდგომში - პროცედურის პირობები) - შესყიდვის ელექტრონული პროცედურის შესახებ განცხადებითა და დოკუმენტაციით გათვალისწინებული ინფორმაცია შესყიდვის შესახებ, აგრეთვე შესყიდვის ობიექტისა და პრეტენდენტის მიმართ კომისიის მიერ დადგენილი პირობების/მოთხოვნების ერთობლიობა;
ზ) პრეტენდენტი - პირი, რომელიც მონაწილეობს შესყიდვის ელექტრონულ პროცედურაში;
თ) მიმწოდებელი - პირი, რომელთანაც დადებულ იქნა ხელშეკრულება შესყიდვის შესახებ;
ი) წინადადების უზრუნველყოფის გარანტია - წინადადების უზრუნველყოფის მიზნით, პრეტენდენტის მიერ შემსყიდველისათვის წარდგენილი გარანტირების მექანიზმი, შესყიდვის ობიექტის სავარაუდო ღირებულების 1%-ის ოდენობით, ხოლო პრეისკურანტის საფუძველზე შესყიდვის შემთხვევაში, სახელშეკრულებო ღირებულების 1%-ის ოდენობით;
კ) წინადადების ფასი - ელექტრონული ვაჭრობის შედეგად დაფიქსირებული წინადადების ფასი, რომელსაც პრეტენდენტი უთითებს სისტემის შესაბამის ველში;
ლ) წინადადების ფასის კლების ბიჯი - შემსყიდველის მიერ შესყიდვის ელექტრონული პროცედურის ჩატარების შესახებ განცხადებაში მითითებული თანხა, შესყიდვის ობიექტის სავარაუდო ღირებულების 0.4%-დან 2%-მდე ოდენობით (შემდგომში - ბიჯი);
მ) წინადადება - ამ წესის მოთხოვნათა დაცვითა და შესყიდვის ელექტრონულ პროცედურაში შემსყიდველის მიერ განსაზღვრული პირობების შესაბამისად, შესყიდვის ელექტრონულ პროცედურაში მონაწილეობის მიღების მიზნით, პრეტენდენტის მიერ წარდგენილი ტექნიკური დოკუმენტაცია და სისტემის მეშვეობით პრეტენდენტის მიერ შეთავაზებული წინადადების ფასი, აგრეთვე, არსებობის შემთხვევაში, შესყიდვის ობიექტის ნიმუში, ექსპერტიზის დასკვნა, ფასწარმოქმნის ადეკვატურობის დამადასტურებელი დოკუმენტ(ებ)ი ან/და სარეგისტრაციო/უფლებრივი მონაცემების ამსახველი დოკუმენტ(ებ)ი;
ნ) ტექნიკური დოკუმენტაცია - პრეტენდენტის მიერ სისტემაში ატვირთული შესყიდვის ობიექტისა და პრეტენდენტის შესახებ ინფორმაცია, რომელიც მოთხოვნილია შემსყიდველის მიერ განცხადებითა და დოკუმენტაციით;
ო) ელექტრონული ვაჭრობა - ელექტრონული შესყიდვის შემადგენელი პროცედურა, რომლის დროსაც პრეტენდენტს სისტემის საშუალებით შეუძლია, შეამციროს მის მიერ წარდგენილი წინადადების ფასი, შესყიდვის ელექტრონულ პროცედურაში გამარჯვების მოპოვების მიზნით. ელექტრონული ვაჭრობა მოიცავს ძირითად დროსა და დამატებით რაუნდებს, სისტემით განსაზღვრული პირობების შესაბამისად. შემსყიდველის გადაწყვეტილებით, ელექტრონული პროცედურა შესაძლებელია, არ მოიცავდეს ელექტრონული ვაჭრობის დამატებით რაუნდებს;
პ) სარეგისტრაციო/უფლებრივი მონაცემების ამსახველი დოკუმენტი - ადმინისტრაციული ორგანოდან წარსადგენი დოკუმენტი, რომელიც შეიცავს შესყიდვის ელექტრონული პროცედურის პირობებით მოთხოვნილ ინფორმაციას პრეტენდენტის მდგომარეობის/საქმიანობის სამართლებრივ მხარესთან დაკავშირებით (სარეგისტრაციო მონაცემები, უფლებრივი შეზღუდვები, არსებული ფინანსური და სხვა სახის ვალდებულებები და სხვ.);
ჟ) კომისია - შემსყიდველის გადაწყვეტილებით, შემსყიდველის თანამშრომლებისაგან დაკომპლექტებული კომისია, რომელიც ახორციელებს შესყიდვის ელექტრონულ პროცედურას, ასევე უფლებამოსილია, მიიღოს გადაწყვეტილება პირდაპირი შესყიდვის საშუალებით განხორციელების შესახებ, ამ წესითა და საქართველოს კანონმდებლობით დადგენილი წესის შესაბამისად;
რ) შესყიდვის ელექტრონული პროცედურა დასრულებულია უარყოფითი შედეგით - შესყიდვის ელექტრონული პროცედურა მიიჩნევა უარყოფითი შედეგით დასრულებულად, როდესაც შესყიდვის ელექტრონულ პროცედურაში მონაწილე ყველა პრეტენდენტი მიიღებს დისკვალიფიკაციას;
ს) შესყიდვის ელექტრონული პროცედურა არ შედგა - შესყიდვის ელექტრონული პროცედურა მიიჩნევა არშემდგარად, როდესაც არცერთმა პრეტენდენტმა არ მიიღო მონაწილეობა შესყიდვის ელექტრონულ პროცედურაში;
ტ) შესყიდვის ელექტრონული პროცედურა შეწყვეტილია - შესყიდვის ელექტრონული პროცედურა მიიჩნევა შეწყვეტილად, როდესაც შემსყიდველის უფლებამოსილი წარმომადგენლის მიერ აიტვირთება კომისიის შესაბამისი დასაბუთებული სხდომის ოქმი;
უ) შესყიდვის ობიექტის სახელშეკრულებო ღირებულება - ამ წესის შესაბამისად, გამარჯვებულ პრეტენდენტთან დასადები ხელშეკრულების ღირებულება;
ფ) შავი სია - "სახელმწიფო შესყიდვების შესახებ" საქართველოს კანონით გათვალისწინებული შესყიდვებში მონაწილე არაკეთილსინდისიერ პირთა რეესტრი;
ქ) სისტემის "CMR" მოდული - ხელშეკრულების შესრულების მიმდინარეობის თაობაზე ანგარიშის წარდგენისათვის სისტემაში განსაზღვრული მოდული (შემდგომში - CMR ).
მუხლი 3. შემსყიდველის უფლება-მოვალეობები
1. შემსყიდველი უფლებამოსილია:
ა) ამ წესის დაცვით, შეარჩიოს მიმწოდებელი და დადოს მასთან შესყიდვის შესახებ ხელშეკრულება (შემდგომში - ხელშეკრულება);
ბ) ამ წესის შესაბამისად, მოახდინოს პრეტენდენტთა დისკვალიფიკაცია;
გ) ხელშეკრულების დადებამდე ნებისმიერ დროს, დასაბუთებული გადაწყვეტილებით შეწყვიტოს შესყიდვის ელექტრონული პროცედურა, თუ ეს აუცილებელი გახდება მისგან დამოუკიდებელი ან/და წინასწარ გაუთვალისწინებელი ობიექტური მიზეზებით, აგრეთვე შემსყიდველის საქმიანობის მიზნებიდან ან/და ინტერესებიდან გამომდინარე;
დ) განახორციელოს მიმწოდებლის მიერ ხელშეკრულების პირობების შესრულების კონტროლი და ზედამხედველობა;
ე) საქართველოს კანონმდებლობის შესაბამისად, ნებისმიერ დროს შეწყვიტოს ხელშეკრულება, თუ მიმწოდებლის მიერ წარდგენილი სარეგისტრაციო/უფლებრივი მონაცემები მცდარი აღმოჩნდება, აგრეთვე საქართველოს კანონმდებლობითა და მხარეთა შორის არსებული ხელშეკრულებით გათვალისწინებული სხვა შემთხვევების არსებობისას.
2. შემსყიდველი ვალდებულია:
ა) რაციონალურად, მათ შორის, ეკონომიურად, პროდუქტიულად და ეფექტიანად განახორციელოს შესყიდვები, ამ წესით დადგენილი პირობების დაცვით;
ბ) ხელშეკრულებით გათვალისწინებული პირობების შესაბამისად, აუნაზღაუროს მიმწოდებელს მიწოდებული საქონლის ან/და მომსახურების ღირებულება;
გ) შესყიდვის ელექტრონული პროცედურის შეწყვეტის შემთხვევაში, სისტემაში ატვირთოს შესაბამისი გადაწყვეტილების დასაბუთებული სხდომის ოქმი. შესყიდვის ელექტრონული პროცედურის შეწყვეტის შემთხვევაში, შემსყიდველი არ არის ვალდებული, პრეტენდენტებს აუნაზღაუროს შესყიდვაში მონაწილეობასთან ან/და ხელშეკრულების მომზადებასთან დაკავშირებული ხარჯები;
დ) არ დადოს ხელშეკრულება შავ სიაში რეგისტრირებულ მიმწოდებელთან.
მუხლი 4. შესყიდვის საშუალებები
1. ამ წესის დანართით განსაზღვრული საქონლის ან/და მომსახურების შესყიდვა ხორციელდება შესყიდვის ელექტრონული პროცედურის საშუალებით ან/და პირდაპირი შესყიდვის საშუალებით.
2. შესყიდვა შეიძლება განხორციელდეს ერთ პირთან მოლაპარაკების საშუალებით ქართულ ენაზე, ინგლისურ ან სხვა უცხო ენებზე, თუ:
ა) შესყიდვის ობიექტის სავარაუდო ღირებულება არ აღემატება 50 000 (ორმოცდაათი ათასი) ლარს;
ბ) ხორციელდება ამ წესის დანართი №2-ის პირველი ნაწილით გათვალისწინებული მომსახურების შესყიდვა, ღირებულების მაქსიმალური ზღვრული ოდენობის გარეშე;
გ) ხორციელდება ამ წესის დანართი №2-ის მეორე ნაწილით გათვალისიწინებული მომსახურების შესყიდვა და შესყიდვის ობიექტის ღირებულება არ აღემატება 200 000 (ორასი ათასი) ლარს;
მუხლი 5. ინტერესთა კონფლიქტის თავიდან აცილების პირობები
1. ინტერესთა კონფლიქტის თავიდან აცილების პირობები ეხება შესყიდვასთან დაკავშირებულ შემდეგ საქმიანობებს:
ა) შესყიდვის ელექტრონული პროცედურის წინადადების განხილვას, შერჩევასა და შეფასებას;
ბ) ხელშეკრულების შესრულებაზე კონტროლსა და ზედამხედველობას;
გ) მიმწოდებლის შერჩევას პირდაპირი შესყიდვის შემთხვევაში.
2. ამ მუხლის პირველი პუნქტით განსაზღვრული საქმიანობების განმახორციელებელ პირს ექნება ინტერესთა კონფლიქტი პრეტენდენტთან ან მიმწოდებელთან, თუ მათ შორის არსებობს საქართველოს საგადასახადო კოდექსის მე-19 მუხლით გათვალისწინებული ურთიერთობა.
მუხლი 6. შესყიდვების დაგეგმვა
1. ამ წესის დანართით განსაზღვრული საქონლისა და მომსახურების შესყიდვა ხორციელდება ამ წესის შესაბამისად, წინასწარ განსაზღვრული და შემსყიდველის ხელმძღვანელის გადაწყვეტილებით დამტკიცებული გეგმის შესაბამისად.
2. ამ მუხლის პირველი პუნქტით გათვალისწინებული გეგმა რეგისტრირდება სისტემაში და უნდა შეიცავდეს:
ა) შესყიდვის ობიექტის დასახელებას;
ბ) შესყიდვის სავარაუდო ღირებულებას;
გ) შესყიდვის საშუალებას;
დ) შესყიდვის განხორციელების პერიოდს (მითითებას შესაბამის კვარტალზე);
ე) დაფინანსების წყაროს.
3. ყოველი შესყიდვის განხორციელებამდე, სავალდებულოა ბაზრის კვლევის ჩატარება, რომელიც, თავის მხრივ, უნდა განხორციელდეს სახელმწიფო შესყიდვების სფეროში მოქმედი კანონმდებლობის, რეკომენდაციებისა და მეთოდური მითითებების შესაბამისად.
მუხლი 7. კომისიის შექმნისა და საქმიანობის წესი
1. შესყიდვის ელექტრონულ პროცედურას განსაზღვრავს და ატარებს კომისია, რომელსაც არანაკლებ სამი წევრის შემადგენლობით ქმნის შემსყიდველის ხელმძღვანელი.
2. კომისია მოქმედებს შემსყიდველის სახელით.
3. კომისიას თავმჯდომარეობს შემსყიდველის ხელმძღვანელი ან მის მიერ დანიშნული პირი.
4. კომისიის გადაწყვეტილების საფუძველზე, ექსპერტებად და კონსულტანტებად შეიძლება მიწვეულ იქნენ შესაბამისი დარგის სპეციალისტები.
5. კომისია გადაწყვეტილებას იღებს წევრთა სიითი შემადგენლობის უმრავლესობით. ხმათა თანაბრად გაყოფის შემთხვევაში, გადამწყვეტია კომისიის თავმჯდომარის ხმა. კომისიის გადაწყვეტილება აისახება სხდომის ოქმში, რომელსაც ხელს აწერს სხდომაზე დამსწრე კომისიის ყველა წევრი. კომისიის წევრს, რომელიც არ ეთანხმება კომისიის გადაწყვეტილებას, უფლება აქვს, წერილობით წარადგინოს თავისი აზრი, რომელიც უნდა დაერთოს კომისიის გადაწყვეტილებას.
6. კომისიის ნებისმიერი გადაწყვეტილება უნდა აისახოს კომისიის შესაბამის სხდომის ოქმში. ოქმი ატვირთული უნდა იქნეს სისტემაში ხელმოწერისთანავე. სისტემაში ატვირთული ოქმის შეცვლა შესაძლებელია მხოლოდ სისტემაში უკვე ატვირთული ოქმის გაუქმებითა და შეცვლილი ოქმის სისტემაში ატვირთვის გზით.
7. შემსყიდველსა და პრეტენდენტს შორის ნებისმიერი მიმოწერის/წერილობითი შეტყობინების გაგზავნა ხდება სისტემის შესაბამისი მოდულის მეშვეობით.
მუხლი 8. კომისიის აპარატი
1. კომისიის საქმიანობის ტექნიკური და ორგანიზაციული უზრუნველყოფის მიზნით, შემსყიდველის ხელმძღვანელის გადაწყვეტილებით, შემსყიდველის თანამშრომელთა შემადგენლობით იქმნება კომისიის აპარატი (შემდგომში − აპარატი), რომელსაც ხელმძღვანელობს კომისიის თავმჯდომარე.
2. აპარატის ფუნქცია არის შესყიდვის ელექტრონული პროცედურის ჩატარების შესახებ განცხადების, დოკუმენტაციისა და შესყიდვის მიმდინარეობასთან დაკავშირებული ნებისმიერი ორგანიზაციული საკითხის უზრუნველყოფა.
3. აპარატს უფლება აქვს, მიაწოდოს პრეტენდენტებს ინფორმაცია და განმარტებები შესყიდვის ელექტრონულ პროცედურასთან დაკავშირებით.
4. აპარატი ვალდებულია, ყოველი შესყიდვის ელექტრონული პროცედურის ჩატარებასთან დაკავშირებული ნებისმიერი დოკუმენტისა და ინფორმაციული მასალის შენახვა უზრუნველყოს ხელშეკრულების დადებიდან არანაკლებ 10 წლის განმავლობაში.
მუხლი 9. განცხადება შესყიდვის ელექტრონული პროცედურის ჩატარების შესახებ
1. შესყიდვის ელექტრონული პროცედურის ჩატარების შესახებ განცხადების (შემდგომში − განცხადება) შევსება და განთავსება ხდება სისტემის მეშვეობით.
2. განცხადებისა და დოკუმენტაციის გაცნობის ვადა უნდა შეადგენდეს არანაკლებ 3 (სამი) სამუშაო დღეს განცხადებისა და დოკუმენტაციის სისტემაში განთავსების მომენტიდან. ამ ვადის დასრულებიდან არანაკლებ 3 (სამი) დღის განმავლობაში პრეტენდენტს შეუძლია სისტემის საშუალებით წარადგინოს წინადადება და ელექტრონული ვაჭრობის გზით შეცვალოს წარდგენილი წინადადების ფასი კლებადობის პრინციპითა და ბიჯის გათვალისწინებით.
3. განცხადებაში უნდა მიეთითოს შემდეგი ინფორმაცია:
ა) განცხადების ნომერი;
ბ) შემსყიდველის დასახელება;
გ) გამოცხადების თარიღი, ზუსტი დროის მითითებით;
დ) წინადადებების მიღების, დაწყებისა და დასრულების თარიღი, ზუსტი დროის მითითებით;

4. განცხადებას აქვს დამატებითი ინფორმაციის ველი, რომელიც შესაძლებელია გამოყენებულ იქნეს შემსყიდველის უფლებამოსილი წარმომადგენლის მიერ შესყიდვის ობიექტის დამატებითი აღწერის ან შესყიდვის ელექტრონულ პროცედურასთან დაკავშირებული დამატებითი ინფორმაციის მიწოდების მიზნით.
5. განცხადების სისტემაში განთავსებისათვის აუცილებელია დოკუმენტაციის ატვირთვა სისტემაში.
მუხლი 10. შესყიდვის ელექტრონული პროცედურის დოკუმენტაციის სისტემაში განთავსება და დოკუმენტაციაში ცვლილების შეტანა

დ) მეთოდებს, რომლებითაც უნდა იყოს გაანგარიშებული წინადადების ღირებულება იმის მითითებით, უნდა შეიცავდეს თუ არა იგი საქონლის ან მომსახურების ღირებულების გარდა სხვა ხარჯებსაც (ტრანსპორტირება, დაზღვევა, გადასახადები და სხვა);

ვ) ინფორმაციას შესყიდვის ობიექტის საგარანტიო ვადისა და პირობების შესახებ (საჭიროების შემთხვევაში);
ზ) მითითებას წინადადების უზრუნველყოფის 1%-იანი გარანტიის შესახებ (ასეთის არსებობის შემთხვევაში);
თ) კომისიის აპარატის იმ წევრის ვინაობასა და საკონტაქტო ინფორმაციას, რომელსაც უფლება აქვს, გასცეს ინფორმაცია და განმარტებები შესყიდვის ელექტრონული პროცედურის შესახებ.
2. კომისიას უფლება აქვს, წინადადების მიღების დროის დაწყებამდე შეიტანოს ცვლილება განცხადებასა და დოკუმენტაციაში მითითებულ მონაცემებში. დოკუმენტაციაში ცვლილების შეტანა შესაძლებელია დოკუმენტაციის გაუქმებითა და შეცვლილი დოკუმენტაციის სისტემაში ატვირთვის გზით. დაუშვებელია შესყიდვის ობიექტის ცვლილება.
მუხლი 11. ტექნიკური მოთხოვნები პრეტენდენტის მიმართ და შესყიდვის ობიექტის აღწერის წესი
1. შემსყიდველი, ყოველ კონკრეტულ შემთხვევაში, უფლებამოსილია, განსაზღვროს მოთხოვნები, რომლებსაც უნდა აკმაყოფილებდეს პრეტენდენტი და შესყიდვის ობიექტი.
2. პრეტენდენტთა მიმართ მოთხოვნები შესაძლებელია დადგინდეს მათი ცოდნის, გამოცდილების, სანდოობის, ფინანსური და ტექნიკური შესაძლებლობების შეფასების მიზნით.
3. შემსყიდველი უფლებამოსილია, მოითხოვოს სარეგისტრაციო/უფლებრივი მონაცემების ამსახველი დოკუმენტ(ებ)ის წარდგენა, თუ შესყიდვის სავარაუდო ღირებულება შეადგენს ან აღემატება 200 000 ლარს. თუ შესყიდვის სავარაუდო ღირებულება ნაკლებია 200 000 ლარზე, სარეგისტრაციო/უფლებრივი მონაცემების ამსახველი დოკუმენტ(ებ)ის წარდგენის მოთხოვნა დასაშვებია მხოლოდ განსაკუთრებულ შემთხვევებში, რაც უნდა დასაბუთდეს შესყიდვის ელექტრონული პროცედურის პირობებში. ასეთ შემთხვევაში, სარეგისტრაციო/უფლებრივი მონაცემების ამსახველი დოკუმენტების რაოდენობა უნდა იყოს მინიმალური, კონკრეტული შესყიდვის თავისებურებიდან გამომდინარე.
4. კომისია ვალდებულია, შესყიდვის ობიექტის ტექნიკური, შესრულების ან/და ფუნქციური სპეციფიკაციების აღწერისას გამოყენებული ტექსტური მასალები და სხვა მაჩვენებლები შეუსაბამოს საერთაშორისო ან საქართველოს სტანდარტებს. კომისია ვალდებულია, მიუთითოს შესაბამისი გამოყენებული სტანდარტი (არსებობის შემთხვევაში). დაუშვებელია სასაქონლო ნიშნის, პატენტის, მოდელის, წარმოშობის წყაროს ან მწარმოებლის მითითება, გარდა იმ შემთხვევისა, როდესაც შესყიდვის ობიექტის ზუსტი აღწერის სხვა საშუალება არ არსებობს. ასეთ შემთხვევაში, აღწერისას აუცილებლად უნდა იქნეს გამოყენებული ტერმინი "მსგავსი" ან "ეკვივალენტური" და სხვა.
5. შემსყიდველი შესყიდვის ობიექტის ნიმუშის წარდგენის ვალდებულებას და ვადას განსაზღვრავს დოკუმენტაციაში. პრეტენდენტის მიერ ნიმუშის წარუდგენლობა ან მისი შეუსაბამობა განცხადების ან/და დოკუმენტაციის მოთხოვნებთან ან სისტემის მეშვეობით პრეტენდენტის მიერ წარდგენილ ტექნიკურ დოკუმენტაციასთან გამოიწვევს მის დისკვალიფიკაციას, ამ წესის მე-15 მუხლის პირველი პუნქტის "გ" ქვეპუნქტის საფუძველზე. პრეტენდენტის მიერ წარდგენილი შესყიდვის ობიექტის ნიმუში დაზუსტებას არ ექვემდებარება. პრეტენდენტს შესყიდვის ობიექტის ნიმუში უბრუნდება გონივრულ ვადაში, გარდა იმ შემთხვევებისა, როდესაც ნიმუში მალფუჭებადი ნივთია და/ან მისი ბუნებიდან გამომდინარე დაბრუნება შეუძლებელია.
მუხლი 12. წინადადების წარდგენა
1. შესყიდვის ელექტრონულ პროცედურაში მონაწილეობის მისაღებად პრეტენდენტს შეუძლია წარადგინოს წინადადება წინადადების მიღების ვადის დაწყების ეტაპიდან წინადადების მიღების ვადის დასრულებამდე.
2. წინადადების წარდგენამდე ნებისმიერ პირს აქვს უფლება, სისტემაში გაეცნოს განცხადებასა და დოკუმენტაციას.
3. წინადადების წარდგენა შესაძლებელია მხოლოდ სისტემის მეშვეობით. პრეტენდენტის მიერ ატვირთული უნდა იქნეს შესყიდვის ელექტრონული პროცედურების პირობებით განსაზღვრული ყველა დოკუმენტი.
4. ამ მუხლის მე-3 პუნქტით გათვალისწინებული დოკუმენტების ატვირთვის შემდგომ, პრეტენდენტმა სისტემის შესაბამის ველში უნდა მიუთითოს წინადადების ფასი, რომელიც არ უნდა აღემატებოდეს შესყიდვის ობიექტის სავარაუდო ღირებულებას ან/და უნდა იყოს შესყიდვის ობიექტის სავარაუდო ღირებულებაზე, სულ მცირე, ერთი ბიჯით ნაკლები. ტექნიკური დოკუმენტაციის ატვირთვისა და წინადადების ფასის მითითების შემდეგ, დაუშვებელია ატვირთული ტექნიკური დოკუმენტაციის შეცვლა, დამატება, გაუქმება ან წაშლა.
5. იმ შემთხვევაში, თუ განცხადებით გათვალისწინებულია წინადადების ფასის დღგ-ის გარეშე წარდგენა, პრეტენდენტი წინადადების ფასში არ ითვალისწინებს დღგ-ს, მიუხედავად იმისა, არის თუ არა იგი დღგ-ის გადამხდელი, საქართველოს კანონმდებლობის შესაბამისად. ასეთ შემთხვევაში, შერჩევა-შეფასება ხორციელდება წარდგენილი წინადადების ფასების მიხედვით. შემსყიდველი ვალდებულია, ხელშეკრულების ღირებულებაში გაითვალისწინოს წინადადების ფასის დღგ-ის შესაბამისი ოდენობა, თუ ხელშეკრულების დადების მომენტისთვის პრეტენდენტი არის დღგ-ის გადამხდელი, საქართველოს კანონმდებლობის მოთხოვნათა შესაბამისად.
მუხლი 13. წინადადების უზრუნველყოფის გარანტია
1. პრეტენდენტი ვალდებულია ტექნიკურ დოკუმენტაციასთან ერთად წარადგინოს წინადადების უზრუნველყოფის გარანტია, ერთ-ერთი ქვემოთ მითითებული წესითა და ფორმით:
ა) პროცედურის პირობებით განსაზღვრულ საბანკო ანგარიშზე თანხის გადახდითა და აღნიშნულის დამადასტურებელი ქვითრის ტექნიკურ დოკუმენტაციასთან ერთად წარდგენით;
ბ) საბანკო გარანტიით, რომელიც წარდგენილი უნდა იყოს საქართველოს კანონმდებლობის შესაბამისად ლიცენზირებული საბანკო/სადაზღვევო დაწესებულებიდან.
2. წინადადების უზრუნველყოფის გარანტიის ბენეფიციარია შემსყიდველი.
3. თუ შესყიდვის სავარაუდო ღირებულება 50 000 ლარზე ნაკლებია, შემსყიდველი უფლებამოსილია, არ მოითხოვოს წინადადების უზრუნველყოფის გარანტია.
4. წინადადების უზრუნველყოფის გარანტია დაბრუნებას ექვემდებარება შესაბამისი ხელშეკრულების დადების, შესყიდვის ელექტრონული პროცედურის უარყოფითი შედეგით დასრულების ან შეწყვეტის შემთხვევაში, რომელიმე შედეგის დადგომიდან არაუგვიანეს 10 (ათი) სამუშაო დღის განმავლობაში.
5. პრეტენდენტის მიერ წარდგენილი წინადადების უზრუნველყოფის გარანტია არ ექვემდებარება დაბრუნებას, თუ შესაბამისი პრეტენდენტის დისკვალიფიკაცია განხორციელდა ამ წესის მე-15 მუხლის პირველი პუნქტის "ბ", "ვ" ან "ზ" ქვეპუნქტის საფუძველზე.
მუხლი 14. შერჩევა/შეფასება
1. პრეტენდენტის წინადადება ფასდება ელექტრონული ვაჭრობის ეტაპის დასრულების შემდგომ.
2. იმ შემთხვევაში, თუ ორი ან მეტი პრეტენდენტის მიერ სისტემაში მითითებულია წინადადების ყველაზე დაბალი თანაბარი ფასი, უპირატესობა ენიჭება იმ პრეტენდენტს, რომელმაც პირველმა მიუთითა ყველაზე დაბალი წინადადების ფასი.
3. კომისიამ უნდა იმსჯელოს, შეესაბამება თუ არა ყველაზე დაბალი წინადადების ფასის მქონე პრეტენდენტის ტექნიკური დოკუმენტაცია პროცედურის პირობებს. იმ შემთხვევაში, თუ პრეტენდენტის მიერ სისტემაში ატვირთული ტექნიკური დოკუმენტაცია არ შეესაბამება განცხადებასა და დოკუმენტაციაში მითითებულ მოთხოვნებს, კომისიის გადაწყვეტილებით, ხდება პრეტენდენტის დისკვალიფიკაცია, გარდა ამ მუხლის მე-4 პუნქტით გათვალისწინებული შემთხვევებისა.
4. კომისია არ ახდენს პრეტენდენტის დისკვალიფიკაციას, თუ მისი ტექნიკური დოკუმენტაცია არ შეიცავს ისეთ მონაცემ(ებ)ს ან/და შეიცავს ისეთ უზუსტობას/უზუსტობებს, რომლის/რომელთა წარდგენა ან/და დაზუსტება არ გამოიწვევს ტექნიკური დოკუმენტაციის არსებით ცვლილებას ან/და არ გაზრდის წინადადების ფასს.
5. თუ სისტემაში მთლიანი ტექნიკური დოკუმენტაცია ატვირთულია ხარვეზის მქონე ფაილის სახით (რომელიც არ იკითხება), კომისიას უფლება არ აქვს, პრეტენდენტს მიმართოს ტექნიკური დოკუმენტაციის დაზუსტების მოთხოვნით, გარდა იმ შემთხვევისა, როდესაც ხარვეზიანია ტექნიკური დოკუმენტაციის ისეთი ნაწილი, რომლის წარდგენა ან/და დაზუსტება არ გამოიწვევს ტექნიკური დოკუმენტაციის არსებით ცვლილებას ან/და არ გაზრდის წინადადების ფასს.
6. ამ მუხლის მე-4 და მე-5 პუნქტებით გათვალისწინებულ შემთხვევებში, შემსყიდველი მიმართავს პრეტენდენტს მის მიერ წარდგენილი ტექნიკური დოკუმენტაციის დაზუსტების მოთხოვნით, რისთვისაც განუსაზღვრავს გონივრულ ვადას, მაგრამ არაუმეტეს სამი სამუშაო დღისა. თუ აღნიშნულ ვადაში წარდგენილი დოკუმენტაცია არ დაზუსტდება, კომისია ახდენს პრეტენდენტის დისკვალიფიკაციას.
7. იმ შემთხვევაში, თუ წინადადების ყველაზე დაბალი ფასის მქონე პრეტენდენტის მიერ სისტემაში ატვირთული ტექნიკური დოკუმენტაცია, ასევე მოთხოვნის შემთხვევაში სარეგისტრაციო/უფლებრივი მონაცემების ამსახველი დოკუმენტი, შეესაბამება განცხადებასა და დოკუმენტაციაში მითითებულ მოთხოვნებს, შემსყიდველის მიერ ხარჯთაღრიცხვის ან/და ფასების ცხრილის დაზუსტების მოთხოვნა უნდა განხორციელდეს ხელშეკრულების შესრულების უზრუნველყოფის გარანტიის წარდგენის მოთხოვნასა (ასეთის არსებობის შემთხვევაში) და ხელშეკრულების დასადებად მოწვევასთან ერთად.
8. პრეტენდენტის მიერ როგორც სატენდერო წინადადების, ასევე დაზუსტებული ტექნიკური დოკუმენტაციისა და სარეგისტრაციო/უფლებრივი მონაცემების დამადასტურებელი დოკუმენტების წარდგენა შემსყიდველისათვის ხდება სისტემის მეშვეობით.
მუხლი 15. პრეტენდენტის დისკვალიფიკაცია
1. პრეტენდენტი ექვემდებარება დისკვალიფიკაციას, თუ:
ა) იგი შავ სიაშია რეგისტრირებული;
ბ) იგი უარს იტყვის წინადადებაზე ან სხვაგვარად გამოხატავს უარს ხელშეკრულების დადებაზე;
გ) მისი ტექნიკური დოკუმენტაცია არ შეესაბამება ელექტრონული პროცედურის განცხადებითა და დოკუმენტაციით განსაზღვრულ პირობებს;
დ) იგი არ დააზუსტებს ტექნიკურ დოკუმენტაციას დადგენილ ვადაში, არასრულყოფილად დააზუსტებს ან/და დაზუსტების შედეგად არსებითად შეცვლის მას;
ე) იგი არ წარადგენს ნიმუშს, ექსპერტიზის დასკვნას ან/და ფასწარმოქმნის ადეკვატურობის დამადასტურებელ დოკუმენტ(ებ)ს დადგენილ ვადაში ან მის მიერ წარდგენილი ნიმუში, ექსპერტიზის დასკვნა ან/და ფასწარმოქმნის ადეკვატურობის დამადასტურებელი დოკუმენტ(ებ)ი არ შეესაბამება პროცედურის პირობებს;
ვ) იგი არ წარადგენს სარეგისტრაციო/უფლებრივი მონაცემების ამსახველ დოკუმენტ(ებ)ს დადგენილ ვადაში ან/და სარეგისტრაციო/უფლებრივი მონაცემები ან/და სარეგისტრაციო/უფლებრივი მონაცემების ამსახველი დოკუმენტ(ებ)ი არ შეესაბამება პროცედურის პირობებს;
ზ) ხელშეკრულების დადების უფლების მოპოვების მიზნით, ჩაიდენს არაკეთილსინდისიერ ქმედებას.
2. იმ შემთხვევაში, თუ პრეტენდენტს წინადადების ვადის დასრულების დროისთვის არ ექნება მოთხოვნათა სრული დაცვით წარდგენილი წინადადების უზრუნველყოფის გარანტია, პრეტენდენტის შეთავაზება არ განიხილება და კომისიას შეუძლია მიმართოს წინადადების ყველაზე დაბალი ფასის მქონე მომდევნო პრეტენდენტს.
3. ყველაზე დაბალი წინადადების ფასის მქონე პრეტენდენტის დისკვალიფიკაციის შემთხვევაში, კომისია, დადგენილი წესით, აფასებს მომდევნო ყველაზე დაბალი წინადადების ფასის მქონე პრეტენდენტს (არსებობის შემთხვევაში) ან უფლებამოსილია, მიიღოს შესყიდვის ელექტრონული პროცედურის შეწყვეტის შესახებ გადაწყვეტილება, თუ მომდევნო პრეტენდენტის წინადადების ფასი, ბაზარზე შეცვლილი გარემოებების შედეგად, მნიშვნელოვნად აღემატება ამ შესყიდვის ობიექტის საბაზრო ღირებულებას.
მუხლი 16. ხელშეკრულება და სისტემაში ასატვირთი დოკუმენტაცია
1. ამ წესის თანახმად, როგორც ელექტრონული პროცედურის, ასევე პირდაპირი შესყიდვის შედეგად განხორციელებული შესყიდვები, დადებული ხელშეკრულებები და მასთან დაკავშირებული ფაქტობრივი გადახდებისა და ხელშეკრულების შესრულების მიმდინარეობის თაობაზე ინფორმაცია ან/და დოკუმენტები იტვირთება სისტემაში, მათი შედგენიდან არაუგვიანეს 10 დღისა.
2. პირდაპირი შესყიდვის საფუძველზე დადებული ხელშეკრულება, ხელშეკრულების დადებასთან დაკავშირებული დოკუმენტები, ამ ხელშეკრულების ცვლილების ამსახველი დოკუმენტ(ებ)ი, აგრეთვე ხელშეკრულების შესრულების/შეუსრულებლობის დამადასტურებელი დოკუმენტაცია იტვირთება სისტემის CMR მოდულის შესაბამის ველში.
3. ამ წესის შესაბამისად, შესყიდვის ელექტრონულ პროცედურაში გამარჯვებული პრეტენდენტის წინადადებით განსაზღვრული პირობებისა და დოკუმენტაციის საფუძველზე, გამარჯვებულ პრეტენდენტსა და შემსყიდველს შორის იდება ხელშეკრულება, რომელიც ატვირთული უნდა იქნეს სისტემაში.
4. ხელშეკრულების ღირებულება არ შეიძლება აღემატებოდეს გამარჯვებული პრეტენდენტის მიერ ელექტრონული ვაჭრობის შედეგად დაფიქსირებულ წინადადების საბოლოო ფასს, გარდა ამ წესის მე-12 მუხლის მე-5 პუნქტით გათვალისწინებული შემთხვევისა, როდესაც შემსყიდველი ხელშეკრულების ღირებულებაში ითვალისწინებს წინადადების ფასის შესაბამისი დღგ-ის ოდენობას.
5. ხელშეკრულება შესაძლებელია დაიდოს როგორც ქართულ, ასევე უცხოურ ენაზე. ხელშეკრულების უცხოურ ენაზე დადების შემთხვევაში, იგი, საქართველოს კანონმდებლობით დადგენილი წესით, უნდა ითარგმნოს ქართულ ენაზე. ხელშეკრულებას არ სჭირდება თარგმნა, თუ იგი იდება ერთ დოკუმენტად პარალელურად ქართულ და უცხოურ ენებზე.
6. თუ ხელშეკრულების ღირებულება შეადგენს ან აღემატება 200 000 ლარს, შემსყიდველი უფლებამოსილია, მოითხოვოს ხელშეკრულების შესრულების უზრუნველყოფის გარანტია, ხელშეკრულების ღირებულების 2%-დან 10%-მდე ოდენობით.
7. თუ შესყიდვის ელექტრონულ პროცედურაში გამარჯვებული პრეტენდენტის მიერ ელექტრონული ვაჭრობის შედეგად დაფიქსირებული წინადადების საბოლოო ფასი ან პირდაპირი შესყიდვისას დასადები ხელშეკრულების ღირებულება ნაკლებია 200 000 ლარზე, დაუშვებელია, პრეტენდენტს/მიმწოდებელს მოეთხოვოს ხელშეკრულების შესრულების უზრუნველყოფის გარანტიის წარდგენა, გარდა იმ შემთხვევისა, როდესაც შემსყიდველის მიერ დასაბუთებულია, თუ შესყიდვის ობიექტის რა თავისებურებამ განაპირობა გარანტიის მოთხოვნა.
8. შესყიდვის ელექტრონული პროცედურის შედეგად დადებული ხელშეკრულების შესრულების უზრუნველყოფის გარანტია წარდგენილი უნდა იქნეს ხელშეკრულების დადებამდე, ხოლო პირდაპირი შესყიდვისას ხელშეკრულების შესრულების უზრუნველყოფის გარანტია შესაძლებელია შემსყიდველს წარედგინოს ხელშეკრულების დადების შემდეგ, არაუგვიანეს 10 დღისა. გარანტიის წარუდგენლობის შემთხვევაში შემსყიდველი უფლებამოსილია შეწყვიტოს ხელშეკრულება.
9. წინასწარი ანგარიშსწორების შემთხვევაში, მიმწოდებელი ვალდებულია, შემსყიდველს წარუდგინოს წინასწარ გადასახდელი თანხის იდენტური ოდენობის გარანტია, ხელშეკრულებით გათვალისწინებული პირობებით.
10. მიმწოდებელთან დადებული ხელშეკრულების პირობების გადასინჯვის წესი განისაზღვრება საქართველოს კანონმდებლობით.
11. ხელშეკრულების დადება არ არის სავალდებულო ინტერნეტით გამოწერის გზით განსახორციელებელი პირდაპირი შესყიდვის შემთხვევაში, რა დროსაც ხელშეკრულება შესაძლებელია ჩანაცვლდეს ინტერნეტის საშუალებით გადახდის დამადასტურებელი დოკუმენტით.
მუხლი 17. ელექტრონული პროცედურის ჩატარება პრეისკურანტის საფუძველზე
1. ელექტრონული პროცედურის მეშვეობით შესყიდვა შესაძლებელია განხორციელდეს პრეისკურანტის საფუძველზე.
2. ამ მუხლის პირველი პუნქტით გათვალისწინებულ შემთხვევაში, შემსყიდველი ელექტრონული პროცედურის განცხადების პრეისკურანტის სავარაუდო ღირებულების ველში უთითებს პრეისკურანტის სავარაუდო ღირებულებას, ხოლო შესყიდვის ობიექტის სახელშეკრულებო ღირებულების ველში - შესყიდვის ობიექტის სახელშეკრულებო ღირებულებას, რომელიც შესაძლოა განსხვავდებოდეს პრეისკურანტის ღირებულებისაგან.
3. პრეტენდენტის მიერ სისტემის მეშვეობით წარდგენილი წინადადების ფასი არ უნდა აღემატებოდეს პრეისკურანტის სავარაუდო ღირებულებას და ელექტრონული ვაჭრობის შედეგად წინადადების ფასის კლება შესაძლებელია, განხორციელდეს მხოლოდ პრეისკურანტის სავარაუდო ღირებულებასთან მიმართებით.
4. პრეისკურანტის საფუძველზე ელექტრონული პროცედურის ჩატარებისას, კომისია ელექტრონული პროცედურის დოკუმენტაციაში თვითონ განსაზღვრავს პრეისკურანტში მოცემული შესყიდვის ობიექტის ერთეულების ღირებულების დადგენის წესს გამარჯვებული პრეტენდენტის მიერ სისტემის მეშვეობით წარდგენილი წინადადების საბოლოო ფასთან მიმართებით.
5. ამ მუხლით გათვალისწინებულ შემთხვევაში, ხელშეკრულების დადების შემდეგ პრეისკურანტში შესყიდვის ობიექტის რომელიმე ერთეულის არარსებობისას, აღნიშნული ერთეულის ფასი მისი შესყიდვის აუცილებლობისას დგინდება შემსყიდველისა და მიმწოდებლის შეთანხმებით, საბაზრო ფასის გათვალისწინებით. ასეთ შემთხვევაში, პრეისკურანტში შესაბამისი ერთეულის არარსებობით გამოწვეული შესყიდვის ხვედრითი წილი არ შეიძლება, აღემატებოდეს მიმწოდებლის მიერ ელექტრონული ვაჭრობის შედეგად დაფიქსირებული პრეისკურანტის საბოლოო ჯამური ღირებულების 10%-ს.
დანართი №1
სს საქართველოს ენერგეტიკის განვითარების ფონდის საქმიანობის სპეციფიკასთან დაკავშირებული საქონლის ჩამონათვალი
№
საქონლის დასახელება
1
ქარის მეტეოანძის, ელექტროსადგურისა და მზის ელექტროსადგურის მოწყობილობები და მათი სათადარიგო ნაწილები, ასევე მათი დემონტაჟ-მონტაჟისათვის საჭირო მასალა-მოწყობილობები.
2
განახლებადი ენერგიის ობიექტების მართვის სისტემა, მისი პროგრამული და არაპროგრამული მოწყობილობები, მისი მონტაჟისათვის საჭირო მასალა-მოწყობილობები.
3
განახლებადი ენერგიის ობიექტების კვლევის, პროექტირებისა და ექსპლუატაციისათვის საჭირო პროგრამა და მისი ლიცენზიები.
დანართი №2
სს საქართველოს ენერგეტიკის განვითარების ფონდის საქმიანობის სპეციფიკასთან დაკავშირებული მომსახურების ჩამონათვალი
№
მომსახურების დასახელება
1
პროექტის ტექნიკურ-ეკონომიკური კვლევა.
2
აუდიტორული მომსახურება.
3
საექსპერტო, საპროექტო, ეკოლოგიის სფეროს, საინჟინრო, არქიტექტურული, საკონსულტაციო, საინვესტიციო, საბანკო, ბაზრის კვლევა და ეკონომიკური კვლევა.
4
განახლებადი ენერგიის ობიექტების კვლევის, პროექტირებისა და ექსპლუატაციისათვის საჭირო პროგრამა და მისი ლიცენზიები.
